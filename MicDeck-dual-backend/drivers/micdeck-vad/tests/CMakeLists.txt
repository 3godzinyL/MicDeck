cmake_minimum_required(VERSION 3.22)
project(MicDeckVadPortableTests LANGUAGES CXX)
enable_testing()

function(micdeck_enable_runtime_assertions target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()

add_library(micdeck_audio_core STATIC ../shared/micdeck_audio_core.cpp ../shared/micdeck_cable_pipeline.cpp)
target_include_directories(micdeck_audio_core PUBLIC ../shared)
target_compile_features(micdeck_audio_core PUBLIC cxx_std_20)
add_executable(core_tests core_tests.cpp)
target_link_libraries(core_tests PRIVATE micdeck_audio_core)
micdeck_enable_runtime_assertions(core_tests)
add_test(NAME core_tests COMMAND core_tests)
add_executable(pipeline_tests pipeline_tests.cpp)
target_link_libraries(pipeline_tests PRIVATE micdeck_audio_core)
micdeck_enable_runtime_assertions(pipeline_tests)
add_test(NAME pipeline_tests COMMAND pipeline_tests)
